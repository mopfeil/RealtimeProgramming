\lstdefinestyle{arduino}{
  language=C++,
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=\color{blue!70!black},
  commentstyle=\color{green!40!black},
  numbers=left, numberstyle=\tiny\color{gray},
  breaklines=true, columns=fullflexible, keepspaces=true,
  frame=single, rulecolor=\color{gray!50},
}
\lstset{style=arduino}

\newcommand{\scriptonly}[1]{\iftoggle{slides}{}{#1}}

  \lstinputlisting[caption={#2},label={#3}]{examples/#1/sketch.ino}%
}

% 8051 assembler (EdSim51 examples)
\lstdefinelanguage{asm51}{
  sensitive=false,
  morecomment=[l]{;},
  morekeywords={MOV,PUSH,POP,INC,DEC,ADD,SUBB,CLR,SETB,CPL,JMP,CALL,RET,RETI,
    JB,JNB,JC,JNC,JZ,JNZ,DJNZ,SJMP,LJMP,LCALL,ORG,EQU,END,
    SUB,CMP,JA,JNA,MUL,DIV,HLT,DB},
}
\lstdefinestyle{asm51}{style=arduino, language=asm51, basicstyle=\ttfamily\scriptsize}

% EdSim51 example: examples/<dir>/<file>.asm
\newcommand{\edsimexample}[4]{% dir, file, caption, label
  \lstinputlisting[style=asm51,caption={#3},label={#4}]{examples/#1/#2.asm}%
}
